Gate turn_b(const Slab& a, const Probe& b) {
    if (!b.skipping) {
        return {true, "full"};
    }
    if (b.op == Op::All) {
        return {true, "all"};
    }
    if (b.op == Op::IsNull) {
        if (a.generation < 3) {
            return {true, "legacy-null"};
        }
        return {a.has_absent, a.has_absent ? "possible" : "empty"};
    }
    if (b.op == Op::RegionEqual) {
        const bool read = b.text >= a.region_low && b.text <= a.region_high;
        return {read, read ? "text-hit" : "text-away"};
    }

    if (a.generation < 3) {
        return {true, "legacy-range"};
    }

    const std::int64_t left = b.x;
    const std::int64_t right = b.op == Op::Equal ? b.x : b.y;
    const bool read = !(right < a.low || left > a.high);
    return {read, read ? "range-hit" : "range-away"};
}

bool is_live(std::uint8_t present, std::uint8_t origin) {
    const auto decoded = static_cast<Origin>(origin);
    if (decoded == Origin::Retired) {
        return false;
    }
    if (decoded == Origin::Inherited) {
        return true;
    }
    return present != 0;
}

pub(crate) fn tilt_a(a: &[i64], b: &[u8], c: &[u8]) -> Result<Mark, FoldErr> {
    if a.is_empty() {
        return Err(FoldErr::Empty);
    }
    if a.len() != b.len() || a.len() != c.len() {
        return Err(FoldErr::Shape);
    }

    let mut low: Option<i64> = None;
    let mut high: Option<i64> = None;
    let mut has_absent = false;
    for ((value, present), origin) in a.iter().zip(b.iter()).zip(c.iter()) {
        if is_live(*present, *origin) {
            low = Some(low.map_or(*value, |current| current.min(*value)));
            high = Some(high.map_or(*value, |current| current.max(*value)));
        } else {
            has_absent = true;
        }
    }

    Ok(Mark {
        low: low.unwrap_or(0),
        high: high.unwrap_or(0),
        has_absent,
    })
}
